-- 1. USER
-- -------------------------------------------------------
CREATE TABLE "USER" (
    USER_ID    SERIAL PRIMARY KEY,
    EMAIL      VARCHAR(50) UNIQUE NOT NULL,
    PASSWORD   VARCHAR(50) NOT NULL,
    ROLE       VARCHAR(50),
    IS_ACTIVE  BOOLEAN DEFAULT TRUE,
    CREATED_AT DATE DEFAULT CURRENT_DATE
);

-- -------------------------------------------------------
-- 2. COMPLIANCE_OFFICER
-- -------------------------------------------------------
CREATE TABLE COMPLIANCE_OFFICER (
    OFFICER_ID    SERIAL PRIMARY KEY,
    EMPLOYEE_ID   VARCHAR(50) UNIQUE NOT NULL,
    DEPARTMENT    VARCHAR(50),
    USER_ID       INT REFERENCES "USER"(USER_ID) ON DELETE SET NULL
);

-- -------------------------------------------------------
-- 3. INVESTOR
-- -------------------------------------------------------
CREATE TABLE INVESTOR (
    INVESTOR_ID   SERIAL PRIMARY KEY,
    USER_ID       INT REFERENCES "USER"(USER_ID) ON DELETE SET NULL,
    FULL_NAME     VARCHAR(50) NOT NULL,
    EMAIL_ID      VARCHAR(50),
    PAN_NO        CHAR(10)  UNIQUE,
    MOBILE_NO     VARCHAR(15),
    DOB           DATE,
    AGE           INT,
    AADHAR_NO     CHAR(12)  UNIQUE,
    KYC_STATUS    VARCHAR(30),
    CREATED_AT    DATE DEFAULT CURRENT_DATE,
    INVESTOR_TYPE VARCHAR(20)
        CHECK (INVESTOR_TYPE IN ('RETAIL', 'HNI', 'INSTITUTIONAL'))
);

-- -------------------------------------------------------
-- 4. KYC_DOCUMENT  (Investor SUBMITS; Officer VERIFIES)
-- -------------------------------------------------------
CREATE TABLE KYC_DOCUMENT (
    DOC_ID              SERIAL PRIMARY KEY,
    DOC_NO              VARCHAR(50) UNIQUE NOT NULL,
    DOC_TYPE            VARCHAR(20)
        CHECK (DOC_TYPE IN ('PAN', 'AADHAR', 'PASSPORT', 'DRIVING_LICENSE')),
    DOC_URL             TEXT,
    ISSUE_DATE          DATE,
    EXPIRY_DATE         DATE,
    VERIFICATION_STATUS VARCHAR(20)
        CHECK (VERIFICATION_STATUS IN ('PENDING', 'VERIFIED', 'REJECTED')),
    OFFICER_ID          INT REFERENCES COMPLIANCE_OFFICER(OFFICER_ID) ON DELETE SET NULL,
    INVESTOR_ID         INT REFERENCES INVESTOR(INVESTOR_ID) ON DELETE CASCADE,
    VERIFIED_BY         VARCHAR(50)
);

-- -------------------------------------------------------
-- 19. ORDER_RECORD  (Investor / TradingAcc PLACES)
-- -------------------------------------------------------
CREATE TABLE ORDER_RECORD (
    ORDER_ID        SERIAL PRIMARY KEY,
    INVESTOR_ID     INT REFERENCES INVESTOR(INVESTOR_ID) ON DELETE SET NULL,
    SECURITY_ID     INT REFERENCES SECURITY(SECURITY_ID) ON DELETE SET NULL,
    TA_ID           INT REFERENCES TRADING_ACC(TA_ID) ON DELETE SET NULL,
    PLACED_BY_BROKER INT REFERENCES BROKER(BROKER_ID) ON DELETE SET NULL,
    SIDE            VARCHAR(10)
        CHECK (SIDE IN ('BUY', 'SELL')),
    ORDER_TYPE      VARCHAR(20)
        CHECK (ORDER_TYPE IN ('MARKET', 'LIMIT', 'SL', 'SL-M')),
    PRODUCT_TYPE    VARCHAR(20)
        CHECK (PRODUCT_TYPE IN ('CNC', 'MIS', 'NRML')),
    QUANTITY        NUMERIC(15, 4) NOT NULL,
    LIMIT_PRICE     NUMERIC(15, 4),
    STOP_LOSS_PRICE NUMERIC(15, 4),
    STATUS          VARCHAR(20)
        CHECK (STATUS IN ('OPEN', 'EXECUTED', 'CANCELLED', 'PARTIAL')),
    PLACED_AT       TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UPDATED_AT      TIMESTAMP
);

    STT              NUMERIC(10, 2),
    TOTAL_CHARGES    NUMERIC(10, 2),
